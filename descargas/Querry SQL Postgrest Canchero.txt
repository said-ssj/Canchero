-- 1. Sincronizar Registro de Supabase Auth con usuario y cliente
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
DECLARE
  v_rol_id INT;
  v_nombre TEXT;
  v_telefono TEXT;
  v_rol_nombre TEXT;
BEGIN
  v_nombre := COALESCE(NEW.raw_user_meta_data->>'nombre', NEW.raw_user_meta_data->>'name', 'Usuario');
  v_telefono := COALESCE(NEW.raw_user_meta_data->>'telefono', '');
  -- El rol enviado por el cliente no es confiable; los permisos se asignan después de revisión administrativa.
  v_rol_nombre := 'cliente';

  SELECT id_rol INTO v_rol_id FROM public.rol
  WHERE LOWER(nombre) = LOWER(v_rol_nombre) AND estado = 'A'
  LIMIT 1;
  IF v_rol_id IS NULL THEN
    RAISE EXCEPTION 'El rol cliente no está configurado en public.rol';
  END IF;

  INSERT INTO public.usuario (id_usuario, id_rol, nombre, apellido, correo, password_hash, telefono, usucre, pccre, feccre, estado)
  VALUES (NEW.id, v_rol_id, v_nombre, '', NEW.email, '', v_telefono, 'AUTH_SIGNUP', 'WEBAPP', NOW(), 'A')
  ON CONFLICT (id_usuario) DO NOTHING;

  INSERT INTO public.cliente (id_cliente, id_usuario, tipo_documento, numero_documento, nombres, apellidos, telefono, correo, usucre, pccre, feccre, estado)
  VALUES (NEW.id, NEW.id, 'DNI', COALESCE(NEW.raw_user_meta_data->>'dni', '00000000'), v_nombre, '', v_telefono, NEW.email, 'AUTH_SIGNUP', 'WEBAPP', NOW(), 'A')
  ON CONFLICT (id_cliente) DO NOTHING;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users;
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW
  EXECUTE FUNCTION handle_new_user();

-- 2. Trigger de Auditoría en Actualizaciones (fecmod, pcmod, usmod)
CREATE OR REPLACE FUNCTION trigger_audit_update()
RETURNS TRIGGER AS $$
BEGIN
  NEW.fecmod := NOW();
  NEW.pcmod  := COALESCE(current_setting('app.current_pc', TRUE), 'WEBAPP');
  NEW.usmod  := COALESCE(current_setting('app.current_user', TRUE), 'SYSTEM');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DO $$
DECLARE
  t TEXT;
BEGIN
  FOR t IN
    SELECT unnest(ARRAY[
      'pais', 'departamento', 'provincia', 'distrito',
      'rol', 'privilegio', 'modulo', 'rol_modulo_privilegio', 'usuario', 'bitacora_acceso',
      'sede', 'tipo_cancha', 'cancha',
      'tarifa_cancha', 'tipo_horario',
      'cliente', 'equipo', 'miembro_equipo',
      'plan_suscripcion', 'suscripcion', 'pago_suscripcion',
      'caja', 'sesion_caja', 'tipo_movimiento_caja', 'movimiento_caja', 'arqueo_caja', 'denominacion', 'arqueo_detalle',
      'tipo_implemento', 'implemento', 'alquiler_implemento',
      'categoria_producto', 'producto', 'tipo_movimiento_inv', 'kardex',
      'estado_reserva', 'reserva', 'equipo_partido', 'bloqueo_mantenimiento',
      'metodo_pago', 'venta', 'venta_detalle', 'pago_reserva', 'sesion_pago_resumen'
    ])
  LOOP
    EXECUTE format('
      DROP TRIGGER IF EXISTS trg_audit_update ON %I;
      CREATE TRIGGER trg_audit_update
        BEFORE UPDATE ON %I
        FOR EACH ROW
        EXECUTE FUNCTION trigger_audit_update();
    ', t, t);
  END LOOP;
END $$;

-- 3. Trigger para cálculo de Comisión 5% en pagos de reserva
CREATE OR REPLACE FUNCTION trigger_calcular_comision_pago_reserva()
RETURNS TRIGGER AS $$
BEGIN
  NEW.comision_plataforma := ROUND(NEW.monto * 0.05, 2);
  NEW.monto_neto_dueno    := ROUND(NEW.monto * 0.95, 2);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_comision_pago_reserva ON pago_reserva;
CREATE TRIGGER trg_comision_pago_reserva
  BEFORE INSERT OR UPDATE OF monto ON pago_reserva
  FOR EACH ROW
  EXECUTE FUNCTION trigger_calcular_comision_pago_reserva();

-- 4. Trigger para generar Código de Pase Digital (#CAN-XXXX)
CREATE OR REPLACE FUNCTION trigger_generar_codigo_pase_reserva()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.codigo_pase IS NULL OR NEW.codigo_pase = '' THEN
    NEW.codigo_pase := '#CAN-' || LPAD(NEW.id_reserva::TEXT, 4, '0');
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_codigo_pase_reserva ON reserva;
CREATE TRIGGER trg_codigo_pase_reserva
  BEFORE INSERT ON reserva
  FOR EACH ROW
  EXECUTE FUNCTION trigger_generar_codigo_pase_reserva();
